#include <GL/glut.h>
#include <cmath>

const int WIDTH = 600;
const int HEIGHT = 300;

void drawStar(float centerX, float centerY, float innerRadius, float outerRadius, int points) {
    glBegin(GL_TRIANGLE_FAN);
    glVertex2f(centerX, centerY);
    for (int i = 0; i <= 2 * points; ++i) {
        float angle = M_PI * i / points;
        float r = (i % 2 == 0) ? outerRadius : innerRadius;
        glVertex2f(centerX + r * cos(angle), centerY + r * sin(angle));
    }
    glEnd();
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, 600, 0, 300);

    //Top Green)
    glColor3f(0.0, 106.0/255.0, 78.0/255.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 240);
    glVertex2i(542, 240);
    glVertex2i(542, 300);
    glVertex2i(57, 300);
    glEnd();

    //Yellow
    glColor3f(1.0, 206.0/255.0, 0.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 180);
    glVertex2i(542, 180);
    glVertex2i(542, 240);
    glVertex2i(57, 240);
    glEnd();

    //Green
    glColor3f(0.0, 106.0/255.0, 78.0/255.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 120);
    glVertex2i(542, 120);
    glVertex2i(542, 180);
    glVertex2i(57, 180);
    glEnd();

    //Yellow
    glColor3f(1.0, 206.0/255.0, 0.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 60);
    glVertex2i(542, 60);
    glVertex2i(542, 120);
    glVertex2i(57, 120);
    glEnd();

    // Bottom Green
    glColor3f(0.0, 106.0/255.0, 78.0/255.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 0);
    glVertex2i(542, 0);
    glVertex2i(542, 60);
    glVertex2i(57, 60);
    glEnd();

    //   Red quad
    glColor3f(210.0/255.0, 16.0/255.0, 52.0/255.0);
    glBegin(GL_QUADS);
    glVertex2i(57, 120);
    glVertex2i(237, 120);
    glVertex2i(237, 300);
    glVertex2i(57, 300);
    glEnd();

    //  Star
    glColor3f(1.0, 1.0, 1.0);
    drawStar(147, 210, 15, 35, 5);

    glFlush();
}

int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(WIDTH, HEIGHT);
    glutCreateWindow("Togo Flag");

    glutDisplayFunc(display);
    glClearColor(0.0, 0.0, 0.0, 0.0);

    glutMainLoop();
    return 0;
}
